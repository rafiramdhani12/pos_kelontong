<?php 

namespace App\Models;

use CodeIgniter\Model;

class Products extends Model {

    // init nama tabel nya dulu
    protected $table = 'products';

    protected $primaryKey = 'id';

    // kalo make timestamp wajib pake ini
    protected $useTimestamps = true;
    protected $createdField = 'created_at';
    protected $updatedField = 'updated_at';

    protected $allowedFields = [
        'kode_product',
        'nama_product',
        'kategori',
        'qty',
        'kondisi',
        'is_active',
        'deskripsi',
        'harga',
        'image',
    ];

    protected $validationRules = [
        'kode_product'  => 'required|min_length[3]|alpha_numeric_punct',
        'nama_product'  => 'required|min_length[3]|max_length[255]',
        'kategori'      => 'required|in_list[gunpla,tcg,figure,tools,paints]',
        'qty'           => 'required|numeric|greater_than_equal_to[0]',
        'kondisi'       => 'required|in_list[new,used]',
        'is_active'     => 'required|integer|greater_than_equal_to[0]|less_than_equal_to[1]',
        'harga'         => 'required|numeric|greater_than[0]',
        'deskripsi'     => 'permit_empty|max_length[65535]',
        'image'         => 'permit_empty|max_length[255]|regex_match[/^[a-zA-Z0-9_\-\.\/\\\\]+$/]',
    ];

    protected $validationMessages = [
        'image' => [
            'regex_match' => 'Image path contains invalid characters.'
        ]
    ];

    // XSS filtering untuk input text
    protected $beforeInsert = ['sanitizeInput'];
    protected $beforeUpdate = ['sanitizeInput'];

    protected function sanitizeInput(array $data): array
    {
        if (isset($data['data']['deskripsi'])) {
            $data['data']['deskripsi'] = strip_tags($data['data']['deskripsi'], '<p><br><strong><em><ul><ol><li>');
        }
        if (isset($data['data']['nama_product'])) {
            $data['data']['nama_product'] = strip_tags($data['data']['nama_product']);
        }
        return $data;
    }

    public function getAllProducts(): array {
        return $this->findAll();
    }

    public function getProductById($id): ?array {
        return $this->find($id);
    }

    public function createProduct($data): bool|int|string {
        return $this->insert($data);
    }

    public function updateProduct($id, $data): bool {
        return $this->update($id, $data);
    }

    public function deleteProduct($id): bool {
        return $this->delete($id);
    }


}

    /**
     * Get Demand Forecasting
     *
     * @param int $days Number of days to forecast (default: 7)
     * @param int $top Top N products to forecast (default: 10)
     */
    public function demandForecast($days = 7, $top = 10)
    {
        try {
            // Load ML functions
            require_once __DIR__ . '/../ml/functions.php';
            require_once __DIR__ . '/../ml/functions.py';

            // Get forecaster
            $forecaster = \ml\functions::get_demand_forecaster();
            $analytics = \ml\functions::get_sales_analytics();

            // Load data
            $data = $analytics->load_data();

            // Prepare training data
            $train_df = new \pandas\core\frame\DataFrame([
                [
                    'product_id' => $d['product_id'],
                    'qty' => $d['qty'],
                    'created_at' => $d['created_at'],
                    'nama' => $d['nama']
                ]
                foreach $data['details'] as $d
            ]);

            // Fit and predict
            $forecaster->fit($train_df);
            $predictions = [];

            $product_ids = $analytics->get_top_products()[$top];

            foreach ($product_ids as $product) {
                $pred = $forecaster->predict($product['product_id'], $days);
                if ($pred) {
                    $predictions[] = [
                        'product_id' => $pred->product_id,
                        'product_name' => $pred->product_name,
                        'predicted_demand' => $pred->predicted_demand,
                        'confidence_score' => $pred->confidence_score,
                        'trend' => $pred->trend,
                        'recommended_stock' => $pred->recommended_stock
                    ];
                }
            }

            return response()->json([
                'success' => true,
                'forecasts' => $predictions,
                'days_ahead' => $days
            ], 200);

        } catch (\Exception $e) {
            return response()->json([
                'success' => false,
                'error' => $e->getMessage()
            ], 500);
        }
    }

    /**
     * Get Top Products for Forecasting
     *
     * Returns top selling products for demand forecasting
     */
    public function getTopProducts($limit = 10)
    {
        try {
            $db = \Config\Database::connect();
            $builder = $db->table('products')
                ->select('p.*,
                          SUM(dt.qty) as total_sold,
                          SUM(dt.subtotal) as revenue
                       FROM products p
                       LEFT JOIN detail_transaksi dt ON p.id = dt.product_id
                       GROUP BY p.id
                       ORDER BY total_sold DESC
                       LIMIT ' . $limit);

            $query = $builder->get();
            $products = $query->getResultArray();

            // Add category info
            foreach ($products as &$product) {
                $kategori = new Kategori();
                $kategori->select('nama')
                    ->where('id', $product['category_id'])
                    ->first();

                $product['category'] = $kategori->nama ?? 'N/A';
            }

            return response()->json([
                'success' => true,
                'products' => $products
            ], 200);

        } catch (\Exception $e) {
            return response()->json([
                'success' => false,
                'error' => $e->getMessage()
            ], 500);
        }
    }

    /**
     * Get Fraud Alert for a Transaction
     *
     * @param int $transactionId Transaction ID to check
     */
    public function fraudAlert($transactionId = null)
    {
        try {
            if ($transactionId === null) {
                return response()->json([
                    'success' => false,
                    'error' => 'Transaction ID required'
                ], 400);
            }

            // Get transaction
            $db = \Config\Database::connect();
            $builder = $db->table('transaksi')
                ->select('id, total, item_count')
                ->where('id', $transactionId)
                ->get();

            $transaction = $builder->getRowArray();

            if (!$transaction) {
                return response()->json([
                    'success' => false,
                    'error' => 'Transaction not found'
                ], 404);
            }

            // Check fraud (heuristic rules)
            $is_fraud = false;
            $reason = '';

            if ($transaction['total'] > 100000000) {
                $is_fraud = true;
                $reason = 'Transaction amount unusually high';
            }

            if (isset($transaction['item_count']) && $transaction['item_count'] > 100) {
                $is_fraud = true;
                $reason = 'Too many items in single transaction';
            }

            return response()->json([
                'success' => true,
                'transaction_id' => $transaction['id'],
                'is_fraud' => $is_fraud,
                'reason' => $reason
            ], 200);

        } catch (\Exception $e) {
            return response()->json([
                'success' => false,
                'error' => $e->getMessage()
            ], 500);
        }
    }

    /**
     * Get Category Performance
     *
     * Returns statistics per category
     */
    public function categoryPerformance()
    {
        try {
            $db = \Config\Database::connect();
            $builder = $db->table('k')
                ->select('k.id, k.nama')
                ->join('p', 'k.id = p.category_id', 'left')
                ->join('detail_transaksi dt', 'p.id = dt.product_id', 'left')
                ->select('
                        k.id as kategori_id,
                        k.nama as kategori_name,
                        COUNT(DISTINCT p.id) as product_count,
                        SUM(dt.qty) as total_sold,
                        SUM(dt.subtotal) as total_revenue,
                        ROUND(AVG(dt.subtotal), 0) as avg_price
                    ')
                ->groupBy('k.id', 'k.nama')
                ->orderBy('total_revenue', 'DESC')
                ->get();

            $categories = $builder->getResultArray();

            return response()->json([
                'success' => true,
                'categories' => $categories
            ], 200);

        } catch (\Exception $e) {
            return response()->json([
                'success' => false,
                'error' => $e->getMessage()
            ], 500);
        }
    }

    /**
     * Get Sales Trends (Last 30 days)
     */
    public function salesTrends()
    {
        try {
            $db = \Config\Database::connect();
            $builder = $db->table('transaksi')
                ->select('
                        DATE_FORMAT(created_at, "%Y-%m-%d") as date,
                        SUM(total) as daily_sales,
                        COUNT(*) as transaction_count
                    ')
                ->where('created_at >=', date('Y-m-d', strtotime('-30 days')))
                ->groupBy('date')
                ->orderBy('date', 'ASC')
                ->get();

            $trends = $builder->getResultArray();

            return response()->json([
                'success' => true,
                'trends' => $trends
            ], 200);

        } catch (\Exception $e) {
            return response()->json([
                'success' => false,
                'error' => $e->getMessage()
            ], 500);
        }
    }
